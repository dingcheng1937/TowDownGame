### 2.1 装备基类属性

```gdscript
extends Node2D
class_name BaseEquip

const equip_texture_pre = preload("res://ui/snow/EquipTexture.tscn")

@export var equip_name = ""  # 装备名称

@export var cd_time = 60  # CD冷却时间

var temp_cd = 0  # 当前冷却读秒
var in_equ = false  # 是否装备中
var timer = Timer.new()
var equip_texture_ins  # 状态UI
```

### 2.2 装备使用流程

```gdscript
func _ready():
    timer.wait_time = 0.1
    timer.timeout.connect(on_timeout)
    add_child(timer)

func _input(_event):
    if Input.is_action_pressed(equip_quick_key) && temp_cd <= 0:
        temp_cd = cd_time
        timer.start()
        openFire()

func on_timeout():
    if temp_cd > 0:
        temp_cd -= 0.1
        print(temp_cd)
        if temp_cd == 0:
            timer.stop()
```

### 2.3 装备挂载与卸载

```gdscript
func _enter_tree():
    in_equ = true
    if !equip_texture_ins:
        equip_texture_ins = equip_texture_pre.instantiate()
        equip_texture_ins.name = equip_name
        equip_texture_ins.setEquip(self)
        EquipServer.addUI(equip_texture_ins)

func _exit_tree():
    in_equ = false
    if equip_texture_ins:
        EquipServer.removeUI(equip_texture_ins)
```

### 2.4 装备服务

```gdscript
extends Node

const equip_touch_pre = preload("res://game/equip/EquipOnFloor.tscn")

signal onEquipAdd(ins)
signal onEquipRemove(ins)

func addEquipOnFloor(equip: BaseEquip, position: Vector2):
    var ins = equip_touch_pre.instantiate()
    ins.global_position = position
    ins.equip = equip
    get_tree().call_group("world", "addEquip", ins)

func addEquipToPlayer(equip: BaseEquip):
    Utils.player.addEquip(equip)

func addUI(ins):
    emit_signal("onEquipAdd", ins)

func removeUI(ins):
    emit_signal("onEquipRemove", ins)
```

## 三、代码练习题

### 练习1：实现一个简单的手雷装备

```gdscript
extends BaseEquip

@export var grenade_scene: PackedScene
@export var throw_force = 500.0

func openFire():
    if grenade_scene:
        var grenade = grenade_scene.instantiate()
        grenade.global_position = global_position
        get_tree().root.add_child(grenade)
        
        var direction = (get_global_mouse_position() - global_position).normalized()
        grenade.linear_velocity = direction * throw_force
```

### 练习2：实现被动装备效果

```gdscript
extends BaseEquip

func _enter_tree():
    ._enter_tree()
    # 增加玩家血量上限
    PlayerData.player_hp_max += 20
    PlayerData.player_hp += 20

func _exit_tree():
    ._exit_tree()
    # 移除效果
    PlayerData.player_hp_max -= 20
    if PlayerData.player_hp > PlayerData.player_hp_max:
        PlayerData.player_hp = PlayerData.player_hp_max
```
