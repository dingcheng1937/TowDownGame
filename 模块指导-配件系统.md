## 二、核心代码解析

### 2.1 配件属性

```gdscript
extends Node2D
class_name BaseAttachment

@export var am_id = 0  # 配件ID
@export var am_name = ""  # 配件名称
@export var am_image: Texture  # 配件图片
@export_enum("WEAPON_OPTICS","WEAPON_MUZZLE","WEAPON_BARREL",
            "WEAPON_UNDERBARREL","WEAPON_AMMUNITION",
            "WEAPON_STOCK","WEAPON_TACTICAL","WEAPON_PERKS") 
var am_type = "WEAPON_OPTICS"  # 配件类型
@export_multiline var am_info = ""  # 介绍
@export var money = 20  # 价格
```

### 2.2 武器可用类型限制

```gdscript
@export_group("武器可用类型")
@export var ASSAULT_RIFLES = false
@export var SUBMACHINE_GUNSRELOAD = false
@export var MACHINE_GUNS = false
@export var SNIPER_RIFLES = false
@export var SHOTGUNS = false
@export var LASER_WEAPONS = false

var use_type = []  # 可用类型列表
```

### 2.3 装备到武器流程

```gdscript
# 装备上武器
func onWeaponUp(_gun):
    self.gun = _gun
    onStart()
    gunUpdate()

# 取消武器
func onWeaponDown():
    onDestroy()
    gunUpdate()
    self.gun = null

func gunUpdate():
    gun.updateGun()
```

### 2.4 可用类型检查

```gdscript
# 当前配件是否可安装到枪械
func canUseAm(type):
    return use_type.has(type)

# 可用类型
func _checkTypeList():
    use_type.clear()
    if ASSAULT_RIFLES: use_type.append("ASSAULT_RIFLES")
    if SUBMACHINE_GUNSRELOAD: use_type.append("SUBMACHINE_GUNSRELOAD")
    if MACHINE_GUNS: use_type.append("MACHINE_GUNS")
    if SNIPER_RIFLES: use_type.append("SNIPER_RIFLES")
    if SHOTGUNS: use_type.append("SHOTGUNS")
    if LASER_WEAPONS: use_type.append("LASER_WEAPONS")
```

## 三、代码练习题

### 练习1：实现一个简单的瞄准镜配件

```gdscript
extends BaseAttachment

func onStart():
    # 增加武器的伤害
    if gun:
        gun.damage *= 1.2

func onDestroy():
    # 移除效果
    if gun:
        gun.damage /= 1.2
```

### 练习2：实现配件的视觉效果

```gdscript
extends BaseAttachment

@export var effect_particle: PackedScene
var effect_instance: Node2D

func onStart():
    if effect_particle and gun:
        effect_instance = effect_particle.instantiate()
        gun.add_child(effect_instance)

func onDestroy():
    if effect_instance:
        effect_instance.queue_free()
```
